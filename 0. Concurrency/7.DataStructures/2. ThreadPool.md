
```csharp
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;

public class CustomThreadPool : IDisposable
{
    // The thread-safe queue that holds pending work.
    // It automatically blocks consumer threads when empty.
    private readonly BlockingCollection<Action> _workQueue;
    
    // Keeps track of our long-living worker threads so we can shut them down later.
    private readonly List<Thread> _workers;
    
    private bool _isDisposed;

    /// <summary>
    /// Initializes the Thread Pool and spins up the worker threads.
    /// </summary>
    public CustomThreadPool(int threadCount)
    {
        _workQueue = new BlockingCollection<Action>();
        _workers = new List<Thread>(threadCount);

        for (int i = 0; i < threadCount; i++)
        {
            var thread = new Thread(WorkerLoop)
            {
                // Set to true so these threads don't prevent the application from exiting
                IsBackground = true, 
                Name = $"CustomThreadPool_Worker_{i}"
            };
            
            thread.Start();
            _workers.Add(thread);
        }
    }

    /// <summary>
    /// The Producer method: Adds a new task to the queue.
    /// </summary>
    public void QueueWorkItem(Action workItem)
    {
        if (_isDisposed) 
            throw new ObjectDisposedException(nameof(CustomThreadPool), "Thread pool is shutting down.");
            
        if (workItem == null) 
            throw new ArgumentNullException(nameof(workItem));

        _workQueue.Add(workItem);
    }

    /// <summary>
    /// The Consumer method: This is the infinite loop that each worker thread runs.
    /// </summary>
    private void WorkerLoop()
    {
        // GetConsumingEnumerable() is the magic here.
        // 1. If the queue has items, it yields them immediately.
        // 2. If the queue is empty, it puts the thread to SLEEP (blocks) safely.
        // 3. If CompleteAdding() is called on the queue, the foreach loop gracefully exits.
        foreach (Action workItem in _workQueue.GetConsumingEnumerable())
        {
            try
            {
                // Execute the task
                workItem.Invoke();
            }
            catch (Exception ex)
            {
                // CRITICAL: We must catch exceptions inside the worker loop.
                // If we don't, an unhandled exception in user code will crash and kill our worker thread!
                Console.WriteLine($"[Error in {Thread.CurrentThread.Name}]: {ex.Message}");
            }
        }
        
        Console.WriteLine($"{Thread.CurrentThread.Name} is gracefully shutting down.");
    }

    /// <summary>
    /// Gracefully shuts down the thread pool, allowing running tasks to finish.
    /// </summary>
    public void Dispose()
    {
        if (_isDisposed) return;
        _isDisposed = true;

        // Stop accepting new tasks and signal the worker loops to exit when empty
        _workQueue.CompleteAdding();

        // Wait for all workers to finish their current active tasks
        foreach (var worker in _workers)
        {
            if (worker.IsAlive)
            {
                worker.Join();
            }
        }

        _workQueue.Dispose();
        Console.WriteLine("Thread pool destroyed.");
    }
}
```

### How to use it:

```csharp
public static void Main()
{
    Console.WriteLine("Starting Custom Thread Pool with 3 workers...");
    
    using (var pool = new CustomThreadPool(3))
    {
        // Submit 10 tasks to the pool
        for (int i = 0; i < 10; i++)
        {
            int taskId = i; // Local copy for the closure
            pool.QueueWorkItem(() => 
            {
                Console.WriteLine($"Task {taskId} is being processed by {Thread.CurrentThread.Name}");
                Thread.Sleep(500); // Simulate some work
            });
        }

        // Wait a bit to let them process
        Thread.Sleep(3000);
    } // The 'using' block automatically calls Dispose() here, gracefully shutting down the pool.
}
```

**Deep Dive: Why this design works so well**

 - **Thread Reuse (The Core Goal):** Creating an OS thread is incredibly expensive (it allocates 1MB of memory and requires a kernel transition). By creating the threads once in the constructor and keeping them trapped in the foreach loop, we eliminate thread creation overhead.

 - **The Magic of BlockingCollection:** Before modern .NET, you had to write the WorkerLoop using Monitor.Wait and Monitor.Pulse with a standard `Queue<T>`, carefully managing locks and sleeping states. `BlockingCollection` encapsulates all that complex signaling. If there is no work, the threads cost 0% CPU.

 - **Exception Isolation:** Notice the `try-catch` inside the foreach loop. If a user passes a buggy `Action` that throws a `NullReferenceException`, only that specific task fails. The worker thread catches it, recovers, and immediately moves to the next item in the queue.

 - **Graceful Teardown (Dispose):** When we want to shut down, we don't aggressively kill the threads (remember, `Thread.Abort()` is dangerous). Instead, we call `_workQueue.CompleteAdding()`. This tells the queue, *"Don't accept any more items, and once you are empty, break out of the foreach loops."* We then use `worker.Join()` to wait politely until every thread has packed its bags and exited.
